% =====================================================================
%  Wochenziele-Tabelle
%    \begin{wochenziele}
%      \ziel{Lernpunkt}{Lernziel}
%    \end{wochenziele}
%  Die Nummern werden automatisch gezählt.
% =====================================================================
\newcounter{zielnr}
\newcommand{\ziel}[2]{\stepcounter{zielnr}\thezielnr & #1 & #2 \\}
\newenvironment{wochenziele}{%
  \setcounter{zielnr}{0}%
  \par\noindent
  \renewcommand{\arraystretch}{1.35}%
  \rowcolors{2}{hsluBlau!5}{white}%
  \tabularx{\linewidth}{@{}r>{\raggedright\arraybackslash}p{0.3\linewidth}>{\raggedright\arraybackslash}X@{}}
  \toprule
  \textbf{Nr.} & \textbf{Lernpunkt} & \textbf{Lernziel} \\
  \midrule
}{%
  \bottomrule
  \endtabularx
  \par
}

\tikzset{rechnung/.style={matrix of math nodes, ampersand replacement=\&,
  column sep=2pt, row sep=1pt, nodes={inner sep=1.5pt},
  column 1/.style={anchor=base east}, column 3/.style={anchor=base},
  column 5/.style={anchor=base west}, column 6/.style={anchor=base west}}}
\newcommand{\periode}[3][m]{%
  \draw[-{Latex[length=4pt]}, semithick, pruefFarbe]
    ([yshift=2.5pt]#1-#3-1.base -| #1.west) -- ++(-9pt,0) coordinate (pu)
    |- ([yshift=2.5pt]#1-#2-1.base -| #1.west);
  \node[pruefFarbe, font=\scriptsize, rotate=90, anchor=south]
    at ($(pu)!0.5!(pu |- #1-#2-1.base)$) {Periode};}
